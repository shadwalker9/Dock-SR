cmake_minimum_required(VERSION 3.25)
project(DockSR VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")

option(DOCKSR_ENABLE_VULKAN "Build the standalone Vulkan spatial baseline when tools are available" ON)
option(DOCKSR_REQUIRE_VULKAN "Fail configuration when the Vulkan baseline cannot be built" OFF)
option(BUILD_CLI "Build the standalone scaling and simulation commands" ON)
option(BUILD_PREVIEW "Build the standalone X11 visible preview" OFF)
option(DOCKSR_ENABLE_GATE0_TESTS "Run optional ONNX inspection/CPU-reference tests (requires the Gate-0 Python environment)" OFF)

include(CTest)
find_package(Python3 3.10 REQUIRED COMPONENTS Interpreter)
add_subdirectory(src/core)
if(BUILD_CLI OR BUILD_PREVIEW)
  add_subdirectory(src/image)
endif()

if(DOCKSR_ENABLE_VULKAN AND (BUILD_CLI OR BUILD_PREVIEW))
  find_package(Vulkan QUIET COMPONENTS glslangValidator)
  if(Vulkan_FOUND AND Vulkan_GLSLANG_VALIDATOR_EXECUTABLE)
    add_subdirectory(src/vulkan)
    set(DOCKSR_VULKAN_AVAILABLE ON)
  elseif(DOCKSR_REQUIRE_VULKAN)
    message(FATAL_ERROR "Vulkan headers/loader and glslangValidator are required; see docs/development.md")
  else()
    message(STATUS "Vulkan baseline disabled: install Vulkan headers/loader and glslangValidator (host/mock targets remain available)")
  endif()
elseif(DOCKSR_REQUIRE_VULKAN)
  message(FATAL_ERROR "DOCKSR_REQUIRE_VULKAN needs DOCKSR_ENABLE_VULKAN=ON and BUILD_CLI=ON or BUILD_PREVIEW=ON")
endif()

if(BUILD_PREVIEW)
  find_package(X11 REQUIRED)
  add_subdirectory(src/preview)
endif()

# Copied smoke executables must use the host loader, without preferring a
# developer's absolute SDK prefix. Custom-prefix development activates its
# documented LD_LIBRARY_PATH; no driver/library path is injected by the bundle.
foreach(docksr_command IN ITEMS docksr-scale docksr-preview docksr-bench)
  if(TARGET ${docksr_command})
    set_target_properties(${docksr_command} PROPERTIES SKIP_BUILD_RPATH TRUE)
  endif()
endforeach()

if(BUILD_TESTING)
  add_test(NAME detection COMMAND "${Python3_EXECUTABLE}" "${PROJECT_SOURCE_DIR}/tests/test_detection.py")
  add_test(NAME telemetry COMMAND "${Python3_EXECUTABLE}" "${PROJECT_SOURCE_DIR}/tests/test_telemetry.py")
  add_test(NAME gate0_schemas COMMAND "${Python3_EXECUTABLE}" "${PROJECT_SOURCE_DIR}/tests/test_gate0_schemas.py")
  if(DOCKSR_ENABLE_GATE0_TESTS)
    add_test(NAME gate0_onnx_inspector COMMAND "${Python3_EXECUTABLE}" "${PROJECT_SOURCE_DIR}/tests/test_onnx_inspect.py")
    add_test(NAME gate0_cpu_reference COMMAND "${Python3_EXECUTABLE}" "${PROJECT_SOURCE_DIR}/tests/test_onnx_reference.py")
    set_tests_properties(gate0_onnx_inspector gate0_cpu_reference PROPERTIES LABELS "gate0;host" TIMEOUT 120)
  endif()
  if(BUILD_CLI)
    add_test(NAME docksr-smoke-bundle
             COMMAND "${Python3_EXECUTABLE}" "${PROJECT_SOURCE_DIR}/tests/test_smoke_bundle.py"
                     "${CMAKE_BINARY_DIR}")
    set_tests_properties(docksr-smoke-bundle PROPERTIES LABELS "packaging" TIMEOUT 180)
  endif()
endif()

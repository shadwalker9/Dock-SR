add_library(docksr_image image.cpp)
target_include_directories(docksr_image PUBLIC "${PROJECT_SOURCE_DIR}/include")
target_compile_features(docksr_image PUBLIC cxx_std_20)
if(BUILD_CLI)
    add_executable(docksr-scale main.cpp)
    target_link_libraries(docksr-scale PRIVATE docksr_image docksr_core)
endif()
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(docksr_image PRIVATE -Wall -Wextra -Wpedantic)
    if(BUILD_CLI)
        target_compile_options(docksr-scale PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endif()
if(BUILD_TESTING)
    add_executable(docksr-image-tests "${PROJECT_SOURCE_DIR}/tests/image_tests.cpp")
    target_link_libraries(docksr-image-tests PRIVATE docksr_image)
    add_test(NAME docksr-image COMMAND docksr-image-tests)
    if(BUILD_CLI)
        add_test(NAME docksr-image-cli COMMAND "${Python3_EXECUTABLE}"
                 "${PROJECT_SOURCE_DIR}/tests/test_image_cli.py" "$<TARGET_FILE:docksr-scale>")
        set_tests_properties(docksr-image-cli PROPERTIES TIMEOUT 90)
    endif()
endif()
